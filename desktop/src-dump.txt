import { useState, useEffect } from "react";
import LicenseScreen from "./screens/LicenseScreen";
import ScanningScreen from "./screens/ScanningScreen";
import ResultsScreen from "./screens/ResultsScreen";
import ErrorScreen from "./screens/ErrorScreen";
import { reportSchema } from "./lib/reportSchema";

type AppState = "license" | "scanning" | "results" | "error";

export default function App() {
  const [currentState, setCurrentState] = useState<AppState>("license");
  const [errorMessage, setErrorMessage] = useState("");
  const [reportData, setReportData] = useState<any>(null);

  useEffect(() => {
    window.antigravity.getLicense().then((l: any) => {
      if (l.status === "valid") {
        setReportData(null);
        setCurrentState("results"); 
      }
    });
  }, []);

  const triggerScan = async () => {
    setCurrentState("scanning");
    const res = await window.antigravity.runScan();
    if (res.ok) {
      const parsed = reportSchema.safeParse(res.report);
      if (parsed.success) {
        setReportData(parsed.data);
        setCurrentState("results");
      } else {
        console.error(parsed.error);
        setErrorMessage("The report couldn't be read.");
        setCurrentState("error");
      }
    } else {
      setErrorMessage(res.message);
      setCurrentState("error");
    }
  };

  return (
    <>
      {currentState === "license" && <LicenseScreen onSuccess={triggerScan} />}
      {currentState === "scanning" && <ScanningScreen onCancel={() => { window.antigravity.cancelScan(); setErrorMessage("Scan cancelled."); setCurrentState("error"); }} />}
      {currentState === "results" && (reportData ? <ResultsScreen report={reportData} onScanAgain={triggerScan} /> : 
        <div className="flex items-center justify-center h-screen"><button onClick={triggerScan} className="bg-blue-600 text-white px-6 py-3 rounded shadow">Start Scan</button></div>
      )}
      {currentState === "error" && <ErrorScreen message={errorMessage} onTryAgain={() => setCurrentState("license")} />}
    </>
  );
}

export const BANDS = {
  starting: [0, 4, 1000], host: [4, 8, 1000], backup: [8, 28, 4000], patch: [28, 40, 2000],
  software: [40, 55, 2000], protection: [55, 80, 5000], recovery: [80, 90, 3000], engine: [90, 99, 2000],
} as const;

export function progressPct(phase: keyof typeof BANDS | "done", msInPhase: number): number {
  if (phase === "done") return 100;
  const [s, e, exp] = BANDS[phase];
  return Math.min(e - 1, s + (e - s) * (1 - Math.exp(-msInPhase / exp)));
}

import { useEffect, useState, useRef } from "react";
import { Progress } from "../components/ui/progress";
import { Button } from "../components/ui/button";
import { progressPct, BANDS } from "../lib/progress";
import AdminHint from "../components/AdminHint";

export default function ScanningScreen({ onCancel }: { onCancel: () => void }) {
  const [phase, setPhase] = useState<keyof typeof BANDS>("starting");
  const [pct, setPct] = useState(0);
  const [isAdmin, setIsAdmin] = useState(true);
  const startTime = useRef(Date.now());

  useEffect(() => {
    window.antigravity.appInfo().then((info: any) => setIsAdmin(info.isAdmin));

    const unsub = window.antigravity.onProgress((p: any) => {
      setPhase(p.phase);
      startTime.current = Date.now();
    });
    
    const interval = setInterval(() => setPct(progressPct(phase, Date.now() - startTime.current)), 50);
    return () => { unsub(); clearInterval(interval); };
  }, [phase]);

  return (
    <div className="flex flex-col items-center justify-center min-h-screen p-8 max-w-lg mx-auto text-center">
      <h2 className="text-xl font-bold mb-6 text-rag-grey">Scanning...</h2>
      <Progress value={pct} className="h-3 mb-4 w-full" />
      <p className="mb-8 text-sm font-medium text-rag-grey uppercase tracking-wider">{phase}</p>
      <p className="mb-4 text-gray-500 text-sm">We only read settings. Nothing on your PC is changed.</p>
      
      {!isAdmin && <AdminHint />}
      
      <Button variant="ghost" onClick={onCancel} className="text-red-500">Cancel Scan</Button>
    </div>
  );
}

import { resolve } from 'path'
import { defineConfig, externalizeDepsPlugin } from 'electron-vite'
import type { Plugin } from 'vite'
import react from '@vitejs/plugin-react'
import tailwindcss from '@tailwindcss/vite'

const DEV_CSP =
  "default-src 'self'; script-src 'self' 'unsafe-inline'; style-src 'self' 'unsafe-inline'; " +
  "img-src 'self' data:; connect-src 'self' ws://localhost:* http://localhost:*"

// Dev only: Vite needs an inline preamble + HMR websocket. Packaged build keeps the strict CSP.
const devCsp = (): Plugin => ({
  name: 'ag-dev-csp',
  apply: 'serve',
  transformIndexHtml: {
    order: 'post',
    handler: (html) =>
      html.replace(
        /(<meta http-equiv="Content-Security-Policy" content=")[^"]*(")/,
        `$1${DEV_CSP}$2`
      )
  }
})

export default defineConfig({
  main: { plugins: [externalizeDepsPlugin()] },
  preload: { plugins: [externalizeDepsPlugin()] },
  renderer: {
    resolve: { alias: { '@': resolve(__dirname, 'src/renderer/src') } },
    plugins: [react(), tailwindcss(), devCsp()]
  }
